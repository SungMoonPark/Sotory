### Java - application.properties
```properties
# 애플리케이션 이름 설정
spring.application.name=sotory

# Redis 자동 설정 제외 (직접 설정하려는 경우 사용)
spring.autoconfigure.exclude=org.springframework.boot.autoconfigure.data.redis.RedisAutoConfiguration

# Swagger UI 경로 설정 (/index.html로 Swagger 접속 가능)
springdoc.swagger-ui.path=/index.html

# Swagger API Docs JSON 엔드포인트 경로 설정
springdoc.api-docs.path=/v3/api-docs

# Context path 설정 (주석처리되어 비활성화됨)
#server.servlet.context-path=/api

# 🔐 JWT 설정
# JWT 서명에 사용할 비밀 키
jwt.secret=${JWT_SECRET}

# 액세스 토큰 만료 시간 (ms 단위)
jwt.access-expiration=${JWT_ACCESS_EXPIRATION}

# 리프레시 토큰 만료 시간 (ms 단위)
jwt.refresh-expiration=${JWT_REFRESH_EXPIRATION}

# 리프레시 토큰 갱신 허용 시간 (ms 기준)
jwt.refresh-renew-threshold=${JWT_REFRESH_EXPIRATION_RENEW_THRESHOLD}

# 📡 FastAPI 백엔드 서버와 통신을 위한 기본 URL
fastapi.base.url = ${FASTAPI_URL}

# 🧠 Redis 연결 설정
# Redis 서버 호스트 (기본값: localhost)
spring.data.redis.host=${REDIS_HOST:localhost}

# Redis 서버 포트 (기본값: 6379)
spring.data.redis.port=${REDIS_PORT:6379}

# ☁️ OpenWeather 날씨 API 키 설정
openweather.key=${OPEN_WEATHER_KEY}

# 🐘 PostgreSQL 데이터베이스 설정
# DB 접속 URL (host, port, dbname)
spring.datasource.url=jdbc:postgresql://${DB_HOST:localhost}:${DB_PORT:5432}/${DB_NAME:sotory}

# DB 접속 계정 ID
spring.datasource.username=${DB_USERNAME:root}

# DB 접속 계정 비밀번호
spring.datasource.password=${DB_PASSWORD}

# PostgreSQL JDBC 드라이버 클래스
spring.datasource.driver-class-name=org.postgresql.Driver

# 🛠️ JPA 설정
# DDL 자동 생성 모드 (none이면 스키마 자동 생성 X)
spring.jpa.hibernate.ddl-auto=none

# 실행 쿼리를 콘솔에 출력할지 여부
spring.jpa.show-sql=true

# 출력되는 SQL을 포맷팅해서 보기 좋게 출력
spring.jpa.properties.hibernate.format_sql=true

# Hibernate에서 사용할 DB 방언 지정 (PostgreSQL용)
spring.jpa.properties.hibernate.dialect=org.hibernate.dialect.PostgreSQLDialect

# 💳 핀테크 관련 설정 (외부 API 연동용)
# 핀테크 API 키
fintech.api.key=${FINTECH_API_KEY}

# 핀테크 기관 코드
fintech.institution.code=${FINTECH_INTSTITUTION_CODE}

# 핀테크 앱 번호
fintech.app.no=${FINTECTH_APP_NO}

# 핀테크 기본 URL
fintech.base.url=${FINTECH_BASE_URL}

# 핀테크 관련 추가 설정
fintech.key=${FINTECH_API_KEY}
fintech.regist=${FINTECH_REGISTER_URL}
fintech.user-key=${FINTECH_USER_KEY}

# 로그 설정
# 전체 루트 로그 레벨 INFO로 설정
logging.level.root=INFO

# Redis 로그 레벨 상세 설정 (디버깅용)
logging.level.org.springframework.data.redis=DEBUG
logging.level.io.lettuce.core=DEBUG

# JWT 인증 관련 로그 레벨 설정 (DEBUG로 자세히 보기)
logging.level.com.sotory.auth=DEBUG
logging.level.com.sotory.auth.service=DEBUG

# 사용자 인증에 필요한 API 키
user.api_key=11b5541b-e433-447b-bf0d-56896d08ed49

# 날씨 API 키 재정의 (중복 정의지만 fallback용 가능)
weather.api.key=${OPEN_WEATHER_KEY}
```

    DEFAULT_NEGATIVE_PROMPT: str = (
        "text, letters, writing, logo, watermark, product label, signage, brand, packaging, "
        "menu, advertisement, banner, commercial, barcode, QR code, prices, typographic, printed, stamp"
